#
# Without it the executable has no publisher, description or version in its
# Properties dialog, which is exactly the profile SmartScreen and antivirus
# heuristics treat as suspicious. Keep the four version tuples in sync with
# src/retrocam/__init__.py __version__.

  ffi=FixedFileInfo(
    filevers=(0, 1, 0, 0),
    prodvers=(0, 1, 0, 0),
    mask=0x3F,
    flags=0x0,
    OS=0x40004,
    fileType=0x1,
    subtype=0x0,
    date=(0, 0)
  ),
  kids=[
    StringFileInfo([
      StringTable('040904B0', [
        StringStruct('CompanyName', 'RetroCam Rescue (open source)'),
        StringStruct('FileDescription', 'RetroCam Rescue - rescue photos from old digital cameras'),
        StringStruct('FileVersion', '0.1.0'),
        StringStruct('InternalName', 'retrocam'),
        StringStruct('LegalCopyright', 'MIT License - RetroCam Rescue contributors'),
        StringStruct('OriginalFilename', 'RetroCam Rescue.exe'),
        StringStruct('ProductName', 'RetroCam Rescue'),
        StringStruct('ProductVersion', '0.1.0'),
      ])
    ]),
    VarFileInfo([VarStruct('Translation', [1033, 1200])])
  ]
)
